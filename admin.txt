<html lang="en">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Metosync — Admin Dashboard</title>
<script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
<script src="https://cdn.tailwindcss.com"></script>
<script src="shared.js"></script>
</head>
<body class="bg-[#f4f7f5] min-h-screen">
<div id="app" class="max-w-4xl mx-auto p-4"><p class="text-gray-400 text-center mt-10">Loading…</p></div>
<script>
let session = null;
let myProfile = null;
let myCompany = null;
let errorMsg = '';
let activeTab = 'staff';

function app() { return document.getElementById('app'); }

async function init() {
  const { data } = await sb.auth.getSession();
  session = data.session;
  if (session) await loadProfile();
  render();
}

async function loadProfile() {
  const { data: profile } = await sb.from('admin_users').select('*').eq('id', session.user.id).maybeSingle();
  myProfile = profile;
  if (myProfile) {
    const { data: company } = await sb.from('companies').select('*').eq('id', myProfile.company_id).maybeSingle();
    myCompany = company;
  }
}

function render() {
  if (!session) return renderLogin();
  if (!myProfile) return renderNoCompany();
  renderDashboard();
}

function renderLogin() {
  app().innerHTML = `
    <div class="text-center mt-10 mb-6">
      <img src="metosync-logo.png" class="h-20 mx-auto mb-3 drop-shadow-lg" alt="Metosync">
      <h1 class="text-xl font-bold text-[#0d2a4d]">Company Admin Login</h1>
    </div>
    <div class="bg-white rounded-2xl p-6 shadow-sm max-w-sm mx-auto">
      <input id="email" type="email" autocomplete="off" placeholder="Email" class="w-full border rounded-xl px-4 py-3 mb-3">
      <div class="relative mb-3">
        <input id="password" type="password" autocomplete="off" placeholder="Password" class="w-full border rounded-xl px-4 py-3 pr-10">
        <button type="button" onclick="toggleEye('password',this)" class="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400">👁</button>
      </div>
      ${errorMsg ? `<p class="text-red-600 text-sm mb-3">${errorMsg}</p>` : ''}
      <button id="loginBtn" class="w-full bg-[#0d2a4d] text-white rounded-xl py-3 font-medium">Log In</button>
    </div>`;
  document.getElementById('loginBtn').onclick = async () => {
    errorMsg = '';
    const email = document.getElementById('email').value.trim();
    const password = document.getElementById('password').value;
    const { data, error } = await sb.auth.signInWithPassword({ email, password });
    if (error) { errorMsg = error.message; render(); return; }
    session = data.session;
    await loadProfile();
    render();
  };
}

function renderNoCompany() {
  app().innerHTML = `
    <div class="bg-white rounded-2xl p-6 text-center shadow-sm mt-10 max-w-md mx-auto">
      <p class="text-gray-600 mb-3">This account isn't linked to a company yet.</p>
      <button id="logoutBtn" class="text-sm text-blue-600">Log out</button>
    </div>`;
  document.getElementById('logoutBtn').onclick = async () => { await sb.auth.signOut(); session = null; myProfile = null; render(); };
}

function renderDashboard() {
  const tabs = [
    { id: 'staff', label: 'Staff' },
    { id: 'sites', label: 'Sites' },
    { id: 'attendance', label: 'Attendance' },
    { id: 'payroll', label: 'Payroll' },
  ];
  app().innerHTML = `
    <div class="flex justify-between items-center mb-4 pt-4">
      <div>
        <h1 class="text-lg font-bold text-[#0d2a4d]">Welcome, ${myCompany.name}</h1>
        <p class="text-xs text-gray-400 font-mono">${myCompany.org_code}</p>
      </div>
      <button id="logoutBtn" class="text-sm text-blue-600">Log out</button>
    </div>
    <div class="flex gap-1 mb-4 bg-white rounded-xl p-1 shadow-sm">
      ${tabs.map(t => `<button class="tabBtn flex-1 py-2 rounded-lg text-sm font-medium ${activeTab === t.id ? 'bg-[#0d2a4d] text-white' : 'text-gray-600'}" data-tab="${t.id}">${t.label}</button>`).join('')}
    </div>
    <div class="bg-white rounded-2xl p-6 shadow-sm">
      <p class="text-gray-400 text-center">— ${tabs.find(t => t.id === activeTab).label} content comes next —</p>
    </div>`;
  document.getElementById('logoutBtn').onclick = async () => { await sb.auth.signOut(); session = null; myProfile = null; render(); };
  document.querySelectorAll('.tabBtn').forEach(btn => {
    btn.onclick = () => { activeTab = btn.dataset.tab; render(); };
  });
}

init();
</script>
</body>
</html>
